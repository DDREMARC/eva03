-- EquipaYa: script de la base de datos (parte a)
-- Solo usa lo visto en clase: CREATE DATABASE, USE, CREATE TABLE,
-- PRIMARY KEY, FOREIGN KEY, AUTO_INCREMENT, DEFAULT, INSERT.
-- Las validaciones (dias >= 1, tarifa > 0, documento de 8 u 11
-- digitos) se hacen en Java antes de enviar los datos.
-- ============================================================

CREATE DATABASE IF NOT EXISTS equipaya;
USE equipaya;

-- ------------------------------------------------------------
-- CLIENTE
-- PK = documento (DNI de 8 digitos o RUC de 11). Es texto porque un
-- DNI puede empezar con cero. Al ser PK tiene indice: la consulta por
-- documento (req. 5) no recorre toda la tabla.
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS cliente (
    documento VARCHAR(11)  NOT NULL,
    nombre    VARCHAR(100) NOT NULL,
    telefono  VARCHAR(15),
    PRIMARY KEY (documento)
);

-- ------------------------------------------------------------
-- EQUIPO
-- PK = codigo (texto, el que usa el mostrador).
-- tipo = texto que identifica la clase Java a crear (CAMARA,
-- ILUMINACION, AUDIO). No hay lista fija de tipos en la tabla: agregar
-- DRON el proximo ano no obliga a modificarla (req. 6).
-- disponible = TRUE/FALSE (req. 2).
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS equipo (
    codigo        VARCHAR(10)   NOT NULL,
    descripcion   VARCHAR(100)  NOT NULL,
    tarifa_diaria DECIMAL(10,2) NOT NULL,
    disponible    BOOLEAN       NOT NULL DEFAULT TRUE,
    tipo          VARCHAR(20)   NOT NULL,
    PRIMARY KEY (codigo)
);

-- ------------------------------------------------------------
-- ALQUILER
-- PK = id_alquiler (autoincremental, no hay un dato natural unico:
-- el mismo cliente puede alquilar el mismo equipo muchas veces).
-- Las FK tienen el mismo tipo y tamano que las PK a las que apuntan.
-- monto = lo cobrado, calculado en Java con la regla de cada tipo.
-- devuelto = FALSE mientras el equipo siga fuera (req. 4).
-- ------------------------------------------------------------
CREATE TABLE IF NOT EXISTS alquiler (
    id_alquiler       INT           NOT NULL AUTO_INCREMENT,
    documento_cliente VARCHAR(11)   NOT NULL,
    codigo_equipo     VARCHAR(10)   NOT NULL,
    fecha_inicio      DATE          NOT NULL DEFAULT (CURRENT_DATE),
    dias              INT           NOT NULL,
    monto             DECIMAL(10,2) NOT NULL,
    devuelto          BOOLEAN       NOT NULL DEFAULT FALSE,
    PRIMARY KEY (id_alquiler),
    FOREIGN KEY (documento_cliente) REFERENCES cliente (documento),
    FOREIGN KEY (codigo_equipo)     REFERENCES equipo (codigo)
);

    }

    }

        int resultado = db.marcarAlquilerDevuelto(id);
        if (resultado != 1) {
            return resultado;
        }
        db.liberarEquipo(codigo);
        return 1;
    }

    public int alquilar(String doc, String codigo, int dias, double monto) {
        int resultado = db.marcarEquipoFuera(codigo);
        if (resultado != 1) {
            return resultado;
        }
        int guardado = db.insertarAlquiler(new AlquilerModel(doc, codigo, dias, monto));
        if (guardado != 1) {
            db.liberarEquipo(codigo);
            return -1;
        }
        return 1;
    }

    public ArrayList<PendienteModel> listarPendientes() {
        return db.listarPendientes();
    }

    // Req. 4a
    public ArrayList<FacturacionModel> listarFacturacion() {
        return db.listarFacturacion();
    }
}

import java.util.ArrayList;
import java.util.HashMap;
import java.util.Map;

public class ClientesController {
    private DatabaseConnector db;
    // Req. 5: memoria de clientes ya consultados (clave = documento)
    private Map<String, ClientesModel> memoria = new HashMap<>();

    public ClientesController(DatabaseConnector db) {
        this.db = db;
    }

    // Req. 5: consulta un cliente por documento sin ir a la base cada vez
    public ClientesModel consultarPorDocumento(String documento) {
        if (!documentoValido(documento)) {
            return null;
        }
        if (memoria.containsKey(documento)) {
            return memoria.get(documento);
        }
        ClientesModel clientesModel = db.buscarClientePorDocumento(documento);
        if (clientesModel != null) {
            memoria.put(documento, clientesModel);
        }
        return clientesModel;
    }

    public ArrayList<ClientesModel> listar() {
        return db.listarClientes();
    }

    // Devuelve 1 = registrado, -1 = error de base de datos (ej. documento repetido),
    // -2 = datos invalidos
    public int registrar(String documento, String nombre, String telefono) {
        if (!documentoValido(documento) || nombre.isEmpty()) {
            return -2;
        }
        return db.insertarCliente(new ClientesModel(documento, nombre, telefono));
    }

    // Devuelve 1 = actualizado, 0 = no existe ese documento, -1 = error, -2 = datos invalidos
    public int actualizar(String documento, String nombre, String telefono) {
        if (!documentoValido(documento) || nombre.isEmpty()) {
            return -2;
        }
        int resultado = db.actualizarCliente(new ClientesModel(documento, nombre, telefono));
        if (resultado == 1) {
            memoria.remove(documento);
        }
        return resultado;
    }

    // Req. 3: busca clientes por parte del nombre
    public ArrayList<ClientesModel> buscarPorNombre(String texto) {
        if (texto.isEmpty()) {
            return new ArrayList<>();
        }
        for (int i = 0; i < texto.length(); i++) {
            if (texto.charAt(i) == '%' || texto.charAt(i) == '_') {
                return new ArrayList<>();
            }
        }
        return db.buscarClientes(texto);
    }

    // DNI de 8 digitos o RUC de 11, solo numeros
    private boolean documentoValido(String documento) {
        if (documento.length() != 8 && documento.length() != 11) {
            return false;
        }
        for (int i = 0; i < documento.length(); i++) {
            if (!Character.isDigit(documento.charAt(i))) {
                return false;
            }
        }
        return true;
    }
}

import java.sql.*;
import java.util.ArrayList;

public class DatabaseConnector {
    private String url = "jdbc:mysql://localhost:3306/equipaya";
    private String usuario = "root";
    private String contraseña = "";
    private Connection connection = null;

    public void abrirConexion() {
        if (connection == null) {
            try {
                connection = DriverManager.getConnection(url, usuario, contraseña);
                System.out.println("La conexión ha sido exitosa");
            } catch (SQLException error) {
                System.out.println("Error en la base de datos: " + error.getMessage());
            }
        }
    }

    public int marcarEquipoFuera(String codigo) {
        int filasAfectadas = 0;
        try {
            String query = "UPDATE equipo SET disponible = FALSE WHERE codigo = ? AND disponible = TRUE";
            PreparedStatement ps = connection.prepareStatement(query);
            ps.setString(1, codigo);
            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            return -1;
        }
        return filasAfectadas;
    }

    // Alquiler
    public int insertarAlquiler(AlquilerModel alquilerModel) {
        int filasAfectadas = 0;
        try {
            String sql = "INSERT INTO alquiler (documento_cliente, codigo_equipo, dias, monto) VALUES (?, ?, ?, ?)";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, alquilerModel.getDocCliente());
            ps.setString(2, alquilerModel.getCodEquipo());
            ps.setInt(3, alquilerModel.getDias());
            ps.setDouble(4, alquilerModel.getMonto());

            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error){
            return -1;
        }
        return filasAfectadas;
    }

    public int marcarAlquilerDevuelto(int id) {
        int filasAfectadas = 0;
        try {
            String sql = "UPDATE alquiler SET devuelto = TRUE WHERE id_alquiler = ? AND devuelto = FALSE";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setInt(1, id);

            filasAfectadas = ps.executeUpdate();
            ps.close();

        } catch (SQLException error) {
            return -1;
        }
        return filasAfectadas;
    }

    public int liberarEquipo(String codigo) {
        int filasAfectadas = 0;
        try {
            String sql = "UPDATE equipo SET disponible = TRUE WHERE codigo = ? AND disponible = FALSE";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, codigo);
            filasAfectadas = ps.executeUpdate();
            ps.close();

        } catch (SQLException error) {
            return -1;
        }
        return filasAfectadas;
    }

    public ArrayList<AlquilerModel> listarAlquileres() {
        ArrayList<AlquilerModel> lista = new ArrayList<>();

        try {
            String sql = "SELECT id_alquiler, documento_cliente, codigo_equipo, fecha_inicio, dias, monto, devuelto FROM alquiler";
            Statement st = connection.createStatement();
            ResultSet rs = st.executeQuery(sql);
            while (rs.next()) {
                AlquilerModel alquilerModel = new AlquilerModel(
                        rs.getInt("id_alquiler"),
                        rs.getString("documento_cliente"),
                        rs.getString("codigo_equipo"),
                        rs.getString("fecha_inicio"),
                        rs.getInt("dias"),
                        rs.getDouble("monto"),
                        rs.getBoolean("devuelto"));
                lista.add(alquilerModel);
            }
            rs.close();
            st.close();


        } catch (SQLException error) {
            System.out.println(error.getMessage());
            }
        return lista;
        }

    // Clientes
    public int insertarCliente(ClientesModel cliente) {
        int filasAfectadas = 0;
        try {
            String sql = "INSERT INTO cliente (documento, nombre, telefono) VALUES (?, ?, ?)";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, cliente.getDocumento());
            ps.setString(2, cliente.getNombre());
            ps.setString(3, cliente.getTelefono());
            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            return -1;
        }
        return filasAfectadas;
    }

    public int actualizarCliente(ClientesModel cliente) {
        int filasAfectadas = 0;
        try {
            String sql = "UPDATE cliente SET nombre = ?, telefono = ? WHERE documento = ?";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, cliente.getNombre());
            ps.setString(2, cliente.getTelefono());
            ps.setString(3, cliente.getDocumento());

                if (equipo != null) {
                    lista.add(equipo);
                }
            }

            st.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return lista;

                    + "GROUP BY e.tipo "

            while (rs.next()) {
                FacturacionModel facturacion = new FacturacionModel(

        super(codigo, descripcion, tarifaDiaria, disponible, "CAMARA");
    }

    public double calcularMonto(int dias) {

    }

    public EquipoIluminacion(String codigo, String descripcion, double tarifaDiaria, boolean disponible) {
        super(codigo, descripcion, tarifaDiaria, disponible, "ILUMINACION");

    @Override
    public double calcularMonto(int dias) {
        double subtotal = getTarifaDiaria() * dias;

public class EquipoFabrica {

        public static EquipoModel crear(String tipo, String codigo, String descripcion,
                                        double tarifa, boolean disponible) {
            if (tipo.equals("CAMARA")) {
                return new EquipoCamara(codigo, descripcion, tarifa, disponible);
            } else if (tipo.equals("ILUMINACION")) {
                return new EquipoIluminacion(codigo, descripcion, tarifa, disponible);
            } else if (tipo.equals("AUDIO")) {
                return new EquipoAudio(codigo, descripcion, tarifa, disponible);
            } else if (tipo.equals("DRON")) {
                return new EquipoDron(codigo, descripcion, tarifa, disponible);
            }
            return null;
        }
}
