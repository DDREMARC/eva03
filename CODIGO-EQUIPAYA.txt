import java.util.ArrayList;

public class AlquilerController {
    private DatabaseConnector db;

    public AlquilerController(DatabaseConnector db) {
        this.db = db;
    }

    public ArrayList<AlquilerModel> listar() {
        return db.listarAlquileres();
    }

    public int alquilar(String doc, String codigo, int dias, double monto) {
        int resultado = db.marcarEquipoFuera(codigo);
        if (resultado != 1) {
            return resultado;
        }
        int guardado = db.insertarAlquiler(new AlquilerModel(doc, codigo, dias, monto));
        if (guardado != 1) {
            db.liberarEquipo(codigo);
            return -1;
        }
        return 1;
    }

    public int devolver(int id) {
        String codigo = db.buscarCodigoEquipoPendiente(id);
        if (codigo == null) {
            return 0;
        }
        int resultado = db.marcarAlquilerDevuelto(id);
        if (resultado != 1) {
            return resultado;
        }
        if (db.liberarEquipo(codigo) == -1) {
            db.marcarAlquilerPendiente(id);
            return -1;
        }
        return 1;
    }

    public int revisarConsistencia() {
        ArrayList<EquipoModel> equipos = db.listarEquipos();
        ArrayList<AlquilerModel> alquileres = db.listarAlquileres();
        int corregidos = 0;

        for (int i = 0; i < equipos.size(); i++) {
            EquipoModel equipo = equipos.get(i);
            boolean tienePendiente = false;
            for (int j = 0; j < alquileres.size(); j++) {
                AlquilerModel alquiler = alquileres.get(j);
                if (!alquiler.isDevuelto() && alquiler.getCodEquipo().equals(equipo.getCodigo())) {
                    tienePendiente = true;
                }
            }

            if (tienePendiente && equipo.isDisponible()) {
                if (db.marcarEquipoFuera(equipo.getCodigo()) == 1) {
                    corregidos++;
                }
            } else if (!tienePendiente && !equipo.isDisponible()) {
                if (db.liberarEquipo(equipo.getCodigo()) == 1) {
                    corregidos++;
                }
            }
        }
        return corregidos;
    }

    public ArrayList<PendienteModel> listarPendientes() {
        return db.listarPendientes();
    }

    public ArrayList<FacturacionModel> listarFacturacion() {
        return db.listarFacturacion();
    }
}

import java.util.ArrayList;
import java.util.HashMap;
import java.util.Map;

public class ClientesController {
    private DatabaseConnector db;
    private Map<String, ClientesModel> memoria = new HashMap<>();

    public ClientesController(DatabaseConnector db) {
        this.db = db;
    }

    public ClientesModel consultarPorDocumento(String documento) {
        if (!documentoValido(documento)) {
            return null;
        }
        if (memoria.containsKey(documento)) {
            return memoria.get(documento);
        }
        ClientesModel clientesModel = db.buscarClientePorDocumento(documento);
        if (clientesModel != null) {
            memoria.put(documento, clientesModel);
        }
        return clientesModel;
    }

    public ArrayList<ClientesModel> listar() {
        return db.listarClientes();
    }

    public int registrar(String documento, String nombre, String telefono) {
        if (!documentoValido(documento) || nombre.isEmpty()) {
            return -2;
        }
        return db.insertarCliente(new ClientesModel(documento, nombre, telefono));
    }

    public int actualizar(String documento, String nombre, String telefono) {
        if (!documentoValido(documento) || nombre.isEmpty()) {
            return -2;
        }
        int resultado = db.actualizarCliente(new ClientesModel(documento, nombre, telefono));
        if (resultado == 1) {
            memoria.remove(documento);
        }
        return resultado;
    }

    public ArrayList<ClientesModel> buscarPorNombre(String texto) {
        if (texto.isEmpty()) {
            return new ArrayList<>();
        }
        for (int i = 0; i < texto.length(); i++) {
            if (texto.charAt(i) == '%' || texto.charAt(i) == '_') {
                return new ArrayList<>();
            }
        }
        return db.buscarClientes(texto);
    }

    private boolean documentoValido(String documento) {
        if (documento.length() != 8 && documento.length() != 11) {
            return false;
        }
        for (int i = 0; i < documento.length(); i++) {
            if (!Character.isDigit(documento.charAt(i))) {
                return false;
            }
        }
        return true;
    }
}

// ===== DatabaseConnector.java =====
import java.sql.*;
import java.util.ArrayList;

public class DatabaseConnector {
    private String url = "jdbc:mysql://localhost:3306/equipaya";
    private String usuario = "root";
    private String contraseña = "";
    private Connection connection = null;

    public void abrirConexion() {
        if (connection == null) {
            try {
                connection = DriverManager.getConnection(url, usuario, contraseña);
                System.out.println("La conexión ha sido exitosa");
            } catch (SQLException error) {
                System.out.println("Error en la base de datos: " + error.getMessage());
            }
        }
    }

    public int marcarEquipoFuera(String codigo) {
        int filasAfectadas = 0;
        try {
            String query = "UPDATE equipo SET disponible = FALSE WHERE codigo = ? AND disponible = TRUE";
            PreparedStatement ps = connection.prepareStatement(query);
            ps.setString(1, codigo);
            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
            return -1;
        }
        return filasAfectadas;
    }

    public int insertarAlquiler(AlquilerModel alquilerModel) {
        int filasAfectadas = 0;
        try {
            String sql = "INSERT INTO alquiler (documento_cliente, codigo_equipo, dias, monto) VALUES (?, ?, ?, ?)";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, alquilerModel.getDocCliente());
            ps.setString(2, alquilerModel.getCodEquipo());
            ps.setInt(3, alquilerModel.getDias());
            ps.setDouble(4, alquilerModel.getMonto());

            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
            return -1;
        }
        return filasAfectadas;
    }

    public String buscarCodigoEquipoPendiente(int id) {
        String codigo = null;
        try {
            String sql = "SELECT codigo_equipo FROM alquiler WHERE id_alquiler = ? AND devuelto = FALSE";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setInt(1, id);
            ResultSet rs = ps.executeQuery();
            if (rs.next()) {
                codigo = rs.getString("codigo_equipo");
            }
            rs.close();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
        }
        return codigo;
    }

    public int marcarAlquilerDevuelto(int id) {
        int filasAfectadas = 0;
        try {
            String sql = "UPDATE alquiler SET devuelto = TRUE WHERE id_alquiler = ? AND devuelto = FALSE";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setInt(1, id);

            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
            return -1;
        }
        return filasAfectadas;
    }

    public int marcarAlquilerPendiente(int id) {
        int filasAfectadas = 0;
        try {
            String sql = "UPDATE alquiler SET devuelto = FALSE WHERE id_alquiler = ? AND devuelto = TRUE";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setInt(1, id);

            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
            return -1;
        }
        return filasAfectadas;
    }

    public int liberarEquipo(String codigo) {
        int filasAfectadas = 0;
        try {
            String sql = "UPDATE equipo SET disponible = TRUE WHERE codigo = ? AND disponible = FALSE";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, codigo);
            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
            return -1;
        }
        return filasAfectadas;
    }

    public ArrayList<AlquilerModel> listarAlquileres() {
        ArrayList<AlquilerModel> lista = new ArrayList<>();
        try {
            String sql = "SELECT id_alquiler, documento_cliente, codigo_equipo, fecha_inicio, dias, monto, devuelto FROM alquiler";
            Statement st = connection.createStatement();
            ResultSet rs = st.executeQuery(sql);
            while (rs.next()) {
                AlquilerModel alquilerModel = new AlquilerModel(
                        rs.getInt("id_alquiler"),
                        rs.getString("documento_cliente"),
                        rs.getString("codigo_equipo"),
                        rs.getString("fecha_inicio"),
                        rs.getInt("dias"),
                        rs.getDouble("monto"),
                        rs.getBoolean("devuelto"));
                lista.add(alquilerModel);
            }
            rs.close();
            st.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return lista;
    }

    public int insertarCliente(ClientesModel cliente) {
        int filasAfectadas = 0;
        try {
            String sql = "INSERT INTO cliente (documento, nombre, telefono) VALUES (?, ?, ?)";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, cliente.getDocumento());
            ps.setString(2, cliente.getNombre());
            ps.setString(3, cliente.getTelefono());
            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
            return -1;
        }
        return filasAfectadas;
    }

    public int actualizarCliente(ClientesModel cliente) {
        int filasAfectadas = 0;
        try {
            String sql = "UPDATE cliente SET nombre = ?, telefono = ? WHERE documento = ?";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, cliente.getNombre());
            ps.setString(2, cliente.getTelefono());
            ps.setString(3, cliente.getDocumento());
            filasAfectadas = ps.executeUpdate();
            ps.close();
        } catch (SQLException error) {
            System.out.println("Error en la base de datos: " + error.getMessage());
            return -1;
        }
        return filasAfectadas;
    }

    public ArrayList<ClientesModel> listarClientes() {
        ArrayList<ClientesModel> lista = new ArrayList<>();
        try {
            String sql = "SELECT documento, nombre, telefono FROM cliente";
            Statement st = connection.createStatement();
            ResultSet rs = st.executeQuery(sql);
            while (rs.next()) {
                ClientesModel cliente = new ClientesModel(
                        rs.getString("documento"),
                        rs.getString("nombre"),
                        rs.getString("telefono"));
                lista.add(cliente);
            }
            rs.close();
            st.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return lista;
    }

    public ArrayList<ClientesModel> buscarClientes(String texto) {
        ArrayList<ClientesModel> lista = new ArrayList<>();
        try {
            String sql = "SELECT documento, nombre, telefono FROM cliente WHERE nombre LIKE ?";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, "%" + texto + "%");
            ResultSet rs = ps.executeQuery();
            while (rs.next()) {
                ClientesModel cliente = new ClientesModel(
                        rs.getString("documento"),
                        rs.getString("nombre"),
                        rs.getString("telefono"));
                lista.add(cliente);
            }
            rs.close();
            ps.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return lista;
    }

    public ArrayList<EquipoModel> listarEquipos() {
        ArrayList<EquipoModel> lista = new ArrayList<>();
        try {
            String sql = "SELECT codigo, descripcion, tarifa_diaria, disponible, tipo FROM equipo";
            Statement st = connection.createStatement();
            ResultSet rs = st.executeQuery(sql);
            while (rs.next()) {
                EquipoModel equipo = EquipoFabrica.crear(
                        rs.getString("tipo"),
                        rs.getString("codigo"),
                        rs.getString("descripcion"),
                        rs.getDouble("tarifa_diaria"),
                        rs.getBoolean("disponible"));
                if (equipo != null) {
                    lista.add(equipo);
                }
            }
            rs.close();
            st.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return lista;
    }

    public EquipoModel buscarEquipoPorCodigo(String codigo) {
        EquipoModel equipo = null;
        try {
            String sql = "SELECT codigo, descripcion, tarifa_diaria, disponible, tipo FROM equipo WHERE codigo = ?";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, codigo);
            ResultSet rs = ps.executeQuery();
            if (rs.next()) {
                equipo = EquipoFabrica.crear(
                        rs.getString("tipo"),
                        rs.getString("codigo"),
                        rs.getString("descripcion"),
                        rs.getDouble("tarifa_diaria"),
                        rs.getBoolean("disponible"));
            }
            rs.close();
            ps.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return equipo;
    }

    public ArrayList<PendienteModel> listarPendientes() {
        ArrayList<PendienteModel> lista = new ArrayList<>();
        try {
            String sql = "SELECT c.nombre, e.descripcion, a.dias, a.monto "
                    + "FROM alquiler a "
                    + "JOIN cliente c ON a.documento_cliente = c.documento "
                    + "JOIN equipo e ON a.codigo_equipo = e.codigo "
                    + "WHERE a.devuelto = FALSE";
            Statement st = connection.createStatement();
            ResultSet rs = st.executeQuery(sql);
            while (rs.next()) {
                PendienteModel pendienteModel = new PendienteModel(
                        rs.getString("nombre"),
                        rs.getString("descripcion"),
                        rs.getInt("dias"),
                        rs.getDouble("monto"));
                lista.add(pendienteModel);
            }
            rs.close();
            st.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return lista;
    }

    public ArrayList<FacturacionModel> listarFacturacion() {
        ArrayList<FacturacionModel> lista = new ArrayList<>();
        try {
            String sql = "SELECT e.tipo, SUM(a.monto) AS total "
                    + "FROM alquiler a "
                    + "JOIN equipo e ON a.codigo_equipo = e.codigo "
                    + "GROUP BY e.tipo "
                    + "ORDER BY e.tipo";
            Statement st = connection.createStatement();
            ResultSet rs = st.executeQuery(sql);
            while (rs.next()) {
                FacturacionModel facturacion = new FacturacionModel(
                        rs.getString("tipo"),
                        rs.getDouble("total"));
                lista.add(facturacion);
            }
            rs.close();
            st.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return lista;
    }

    public ClientesModel buscarClientePorDocumento(String documento) {
        ClientesModel cliente = null;
        try {
            String sql = "SELECT documento, nombre, telefono FROM cliente WHERE documento = ?";
            PreparedStatement ps = connection.prepareStatement(sql);
            ps.setString(1, documento);
            ResultSet rs = ps.executeQuery();
            if (rs.next()) {
                cliente = new ClientesModel(
                        rs.getString("documento"),
                        rs.getString("nombre"),
                        rs.getString("telefono"));
            }
            rs.close();
            ps.close();
        } catch (SQLException error) {
            System.out.println(error.getMessage());
        }
        return cliente;
    }

    public Connection getConnection() {
        return connection;
    }

    public void cerrarConexion() {
        if (connection != null) {
            try {
                connection.close();
                connection = null;
            } catch (SQLException error) {
                System.out.println("Error en la base de datos: " + error.getMessage());
            }
        }
    }
}

        int diasCobrados = dias;

        return getTarifaDiaria() * diasCobrados;
    }

public class EquipoFabrica {

    public static EquipoModel crear(String tipo, String codigo, String descripcion,
                                    double tarifa, boolean disponible) {
        if (tipo.equals("CAMARA")) {
            return new EquipoCamara(codigo, descripcion, tarifa, disponible);
        } else if (tipo.equals("ILUMINACION")) {
            return new EquipoIluminacion(codigo, descripcion, tarifa, disponible);
        } else if (tipo.equals("AUDIO")) {
            return new EquipoAudio(codigo, descripcion, tarifa, disponible);
        } else if (tipo.equals("DRON")) {
            return new EquipoDron(codigo, descripcion, tarifa, disponible);
        }
        return null;
    }
}

public class EquipoIluminacion extends EquipoModel {
    private static final double DESCUENTO = 0.10 ;
    public EquipoIluminacion(String codigo, String descripcion, double tarifaDiaria, boolean disponible) {
        super(codigo, descripcion, tarifaDiaria, disponible, "ILUMINACION");
    }

    @Override
    public double calcularMonto(int dias) {
        double subtotal = getTarifaDiaria() * dias;
        if (dias > 5) {
            return subtotal - (subtotal * DESCUENTO);
        } else {
            return subtotal;
        }
    }
}

    }

    public void mostrarInfoFacturacion() {
        System.out.println("Tipo: " + tipo);
        System.out.println("Total facturado: " + total);
    }
}

    }

// ===== MainView.java =====
import java.util.ArrayList;
import java.util.Scanner;

public class MainView {
    private Scanner sc;
    private DatabaseConnector db;
    private ClientesController clientesController;
    private EquipoController equipoController;
    private AlquilerController alquilerController;

    public MainView() {
        sc = new Scanner(System.in);
        db = new DatabaseConnector();
        clientesController = new ClientesController(db);
        equipoController = new EquipoController(db);
        alquilerController = new AlquilerController(db);
    }

    public void iniciar() {
        db.abrirConexion();
        if (db.getConnection() == null) {
            System.out.println("No se pudo conectar a la base de datos. Revisa que MySQL este encendido.");
            sc.close();
            return;
        }

        int corregidos = alquilerController.revisarConsistencia();
        if (corregidos > 0) {
            System.out.println("Se corrigio el estado de " + corregidos + " equipo(s) que no coincidia con sus alquileres.");
        }

        int opcion = -1;
        while (opcion != 0) {
            System.out.println();
            System.out.println("========= EQUIPAYA =========");
            System.out.println("1. Registrar cliente");
            System.out.println("2. Listar clientes");
            System.out.println("3. Modificar cliente");
            System.out.println("4. Listar equipos");
            System.out.println("5. Alquilar equipo");
            System.out.println("6. Devolver equipo");
            System.out.println("7. Listar alquileres");
            System.out.println("8. Buscar cliente por nombre");
            System.out.println("9. Facturado por tipo de equipo");
            System.out.println("10. Alquileres pendientes de devolucion");
            System.out.println("11. Consultar cliente por documento");
            System.out.println("0. Salir");
            opcion = leerEntero("Opcion: ");

            switch (opcion) {
                case 1:
                    registrarCliente();
                    break;
                case 2:
                    listarClientes();
                    break;
                case 3:
                    modificarCliente();
                    break;
                case 4:
                    listarEquipos();
                    break;
                case 5:
                    alquilar();
                    break;
                case 6:
                    devolver();
                    break;
                case 7:
                    listarAlquileres();
                    break;
                case 8:
                    buscarCliente();
                    break;
                case 9:
                    listarFacturacion();
                    break;
                case 10:
                    listarPendientes();
                    break;
                case 11:
                    consultarCliente();
                    break;
                case 0:
                    System.out.println("Saliendo...");
                    break;
                default:
                    System.out.println("Opcion no valida.");
            }
        }

        sc.close();
        db.cerrarConexion();
    }

    private void registrarCliente() {
        System.out.print("Documento (DNI 8 o RUC 11 digitos): ");
        String documento = sc.nextLine();
        System.out.print("Nombre: ");
        String nombre = sc.nextLine();
        System.out.print("Telefono: ");
        String telefono = sc.nextLine();

        int resultado = clientesController.registrar(documento, nombre, telefono);
        if (resultado == 1) {
            System.out.println("Cliente registrado.");
        } else if (resultado == -2) {
            System.out.println("Datos invalidos: revisa el documento y el nombre.");
        } else {
            System.out.println("No se pudo registrar (puede que ese documento ya exista).");
        }
    }

    private void buscarCliente() {
        System.out.print("Parte del nombre: ");
        String texto = sc.nextLine();
        ArrayList<ClientesModel> lista = clientesController.buscarPorNombre(texto);
        if (lista.isEmpty()) {
            System.out.println("No se encontro el cliente.");
        }
        for (int i = 0; i < lista.size(); i++) {
            lista.get(i).mostrarInfo();
            System.out.println("-----");
        }
    }

    private void listarClientes() {
        ArrayList<ClientesModel> lista = clientesController.listar();
        if (lista.isEmpty()) {
            System.out.println("No hay clientes.");
        }
        for (int i = 0; i < lista.size(); i++) {
            lista.get(i).mostrarInfo();
            System.out.println("-----");
        }
    }

    private void modificarCliente() {
        System.out.print("Documento del cliente a modificar: ");
        String documento = sc.nextLine();
        System.out.print("Nuevo nombre: ");
        String nombre = sc.nextLine();
        System.out.print("Nuevo telefono: ");
        String telefono = sc.nextLine();

        int resultado = clientesController.actualizar(documento, nombre, telefono);
        if (resultado == 1) {
            System.out.println("Cliente actualizado.");
        } else if (resultado == 0) {
            System.out.println("No existe un cliente con ese documento.");
        } else if (resultado == -2) {
            System.out.println("Datos invalidos: revisa el documento y el nombre.");
        } else {
            System.out.println("Error en la base de datos.");
        }
    }

    private void listarEquipos() {
        ArrayList<EquipoModel> lista = equipoController.listar();
        if (lista.isEmpty()) {
            System.out.println("No hay equipos.");
        }
        for (int i = 0; i < lista.size(); i++) {
            lista.get(i).mostrarInfo();
            System.out.println("-----");
        }
    }

    private void alquilar() {
        System.out.print("Documento del cliente: ");
        String documento = sc.nextLine();
        System.out.print("Codigo del equipo: ");
        String codigo = sc.nextLine();
        int dias = leerEntero("Dias: ");
        if (dias < 1) {
            System.out.println("Los dias deben ser 1 o mas.");
            return;
        }

        Alquilable equipo = equipoController.buscarPorCodigo(codigo);
        if (equipo == null) {
            System.out.println("No existe un equipo con ese código.");
            return;
        }
        double monto = equipo.calcularMonto(dias);

        int resultado = alquilerController.alquilar(documento, codigo, dias, monto);
        if (resultado == 1) {
            System.out.println("Alquiler registrado. Monto: " + monto);
        } else if (resultado == 0) {
            System.out.println("Ese equipo ya esta fuera.");
        } else {
            System.out.println("No se pudo registrar (revisa que el cliente exista).");
        }
    }

    private void devolver() {
        int id = leerEntero("ID del alquiler: ");

        int resultado = alquilerController.devolver(id);
        if (resultado == 1) {
            System.out.println("Devolucion registrada.");
        } else if (resultado == 0) {
            System.out.println("Ese alquiler ya estaba devuelto o no existe.");
        } else {
            System.out.println("Error en la base de datos.");
        }
    }

    private void listarAlquileres() {
        ArrayList<AlquilerModel> lista = alquilerController.listar();
        if (lista.isEmpty()) {
            System.out.println("No hay alquileres.");
        }
        for (int i = 0; i < lista.size(); i++) {
            lista.get(i).mostrarInfo();
            System.out.println("-----");
        }
    }

    private int leerEntero(String mensaje) {
        System.out.print(mensaje);
        while (!sc.hasNextInt()) {
            System.out.println("Escribe un numero.");
            sc.next();
            System.out.print(mensaje);
        }
        int numero = sc.nextInt();
        sc.nextLine();
        return numero;
    }

    private void listarPendientes() {
        ArrayList<PendienteModel> lista = alquilerController.listarPendientes();
        if (lista.isEmpty()) {
            System.out.println("No hay pendientes.");
        }
        for (int i = 0; i < lista.size(); i++) {
            lista.get(i).mostrarInfo();
            System.out.println("-----");
        }
    }

    private void listarFacturacion() {
        ArrayList<FacturacionModel> lista = alquilerController.listarFacturacion();
        if (lista.isEmpty()) {
            System.out.println("No hay facturacion.");
        }
        for (int i = 0; i < lista.size(); i++) {
            lista.get(i).mostrarInfoFacturacion();
            System.out.println("-----");
        }
    }

    private void consultarCliente() {
        System.out.print("Documento del cliente: ");
        String documento = sc.nextLine();
        ClientesModel clientesModel = clientesController.consultarPorDocumento(documento);

        if (clientesModel == null) {
            System.out.println("El cliente no existe.");
        } else  {
            clientesModel.mostrarInfo();
        }
    }
}


// ===== PendienteModel.java =====
public class PendienteModel {
    private String nombreCliente;
    private String descripcionEquipo;
    private int dias;
    private double monto;

    public PendienteModel(String nombreCliente, String descripcionEquipo, int dias, double monto) {
        this.nombreCliente = nombreCliente;
        this.descripcionEquipo = descripcionEquipo;
        this.dias = dias;
        this.monto = monto;
    }

    public String getNombreCliente() {
        return nombreCliente;
    }

    public String getDescripcionEquipo() {
        return descripcionEquipo;
    }

    public int getDias() {
        return dias;
    }

    public double getMonto() {
        return monto;
    }

    public void mostrarInfo() {
        System.out.println("Cliente: " + nombreCliente);
        System.out.println("Equipo: " + descripcionEquipo);
        System.out.println("Dias: " + dias);
        System.out.println("Monto: " + monto);
    }
}
